  <script>

    // PASTE YOUR PUBLISHED GOOGLE SHEET CSV LINK BELOW

    const SHEET_CSV_URL = "https://docs.google.com/spreadsheets/d/e/2PACX-1vT_-tlrdQlJVk7EAa09PmgqDKVwp2yYcq7e8TykIY76dA8ZFT-qgG2x6deXppicO19DI2q3qi4LmAxz/pub?gid=0&single=true&output=csv";

    function parseCSVRow(row) {

      const result = [];

      let current = "";

      let inQuotes = false;

      for (let i = 0; i < row.length; i++) {

        const char = row[i];

        if (char === '"') {

          inQuotes = !inQuotes;

        } else if (char === "," && !inQuotes) {

          result.push(current.trim());

          current = "";

        } else {

          current += char;

        }

      }

      result.push(current.trim());

      return result;

    }

    async function loadScoreboard() {

      const boardBody = document.getElementById("board-body");

      const totalEl = document.getElementById("total-count");

      const leaderCountEl = document.getElementById("leader-count");

      const leaderNameEl = document.getElementById("leader-name");

      const statusEl = document.getElementById("last-updated");

      try {

        const cacheBuster = `${SHEET_CSV_URL.includes("?") ? "&" : "?"}t=${Date.now()}`;

        const response = await fetch(SHEET_CSV_URL + cacheBuster);

        if (!response.ok) throw new Error("Failed to fetch sheet.");

        const csvText = await response.text();

        const lines = csvText.split(/\r?\n/).filter(line => line.trim() !== "");

        const data = lines.slice(1).map(line => {

          const cols = parseCSVRow(line);

          return {

            committee: cols[0] || "Unknown Committee",

            members: parseInt(cols[1]?.replace(/,/g, ""), 10) || 0

          };

        });

        // Sort strictly in descending order of memberships

        data.sort((a, b) => b.members - a.members);

        const maxMembers = data.length > 0 ? Math.max(...data.map(d => d.members), 1) : 1;

        const totalMembers = data.reduce((sum, item) => sum + item.members, 0);

        totalEl.textContent = totalMembers.toLocaleString();

        if (data.length > 0) {

          leaderCountEl.textContent = data[0].members.toLocaleString();

          leaderNameEl.textContent = `#1 ${data[0].committee}`;

        }

        boardBody.innerHTML = data.map((item, index) => {

          const rank = index + 1;

          const rankDisplay = rank <= 3

            ? `<span class="badge badge-${rank}">${rank}</span>`

            : `<span style="padding-left: 10px; color: #64748B;">${rank}</span>`;

          const barWidth = Math.round((item.members / maxMembers) * 100);

          const rowClass = rank === 1 ? "row rank-first" : "row";

          return `

            <div class="${rowClass}">

              <div class="rank">${rankDisplay}</div>

              <div>

                <div class="committee-name">${item.committee}</div>

                <div class="progress-wrap">

                  <div class="progress-bar" style="width: ${barWidth}%"></div>

                </div>

              </div>

              <div class="count">${item.members.toLocaleString()}</div>

            </div>

          `;

        }).join("");

        const now = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });

        statusEl.textContent = `Live Campaign Feed • Synced at ${now}`;

      } catch (err) {

        boardBody.innerHTML = `<div class="error">Unable to load scoreboard. Make sure your Google Sheet is published to the web as a CSV and the link is added to SHEET_CSV_URL.</div>`;

        statusEl.textContent = "Connection error";

        console.error(err);

      }

    }

    loadScoreboard();

    setInterval(loadScoreboard, 60000);

  </script>
